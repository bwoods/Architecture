10/// A surface, or file format, that views may be rendered to.
11///
12/// ## Note
13/// If a `Dependency::<Transform>` is in scope, the `Output` is expected to
14/// transform its input accordingly.
15#[rustfmt::skip]
16pub trait Output: Sized {
17    /// Sets up the uniform parameters of the following paths.
18    ///
19    /// Path should always begin with a [`move_to`].
20    ///
21    /// [`move_to`]: Self::move_to
22    fn begin(&mut self, rgba: [u8; 4], transform: &Transform);
23
24    /// Begins a new path.
25    ///
26    /// The path should be continued with a series of [`line_to`], [`quadratic_bezier_to`], and/or
27    /// [`cubic_bezier_to`] calls and ended with a call to [`close`].
28    ///
29    /// [`line_to`]: Self::line_to
30    /// [`quadratic_bezier_to`]: Self::quadratic_bezier_to
31    /// [`cubic_bezier_to`]: Self::cubic_bezier_to
32    /// [`close`]: Self::close
33    fn move_to(&mut self, x: f32, y: f32);
34
35    /// Adds a line to the current path.
36    fn line_to(&mut self, x: f32, y: f32);
37
38    /// Adds a quadratic Bézier to the current path.
39    ///
40    /// (`x1`, `y1`) represents the Bézier control point.
41    fn quadratic_bezier_to(&mut self, x1: f32, y1: f32, x: f32, y: f32);
42
43    /// Adds a cubic Bézier to the current path.
44    ///
45    /// (`x1`, `y1`) and (`x2`, `y2`) represent the Bézier control points.
46    fn cubic_bezier_to(&mut self, x1: f32, y1: f32, x2: f32, y2: f32, x: f32, y: f32);
47
48    /// Closes the current path.
49    ///
50    /// Once this method has been called there is no current path until [`begin`] is called again.
51    ///
52    /// [`begin`]: Self::begin
53    fn close(&mut self);
54}
