22/// Warnings logged by `View` modifiers should only be logged once, not on _every_ draw cycle.
23macro_rules! warn_once {
24    ( $( $x:expr ),+ ) => {
25        if cfg!(debug_assertions) {
26            static ONCE: std::sync::Once = std::sync::Once::new();
27            let caller = std::panic::Location::caller();
28
29            ONCE.call_once(|| {
30                tracing::warn!("{}", format!("{}: {}", caller, $( $x ),+));
31            });
32        }
33    };
34}
35#[allow(unused_imports)]
36pub(crate) use warn_once;

46pub trait View: Sized {
47    /// The intrinsic size of the `View`
48    fn size(&self, within: Size) -> Size;
49
50    /// User-interface [`Event`] handling of the `View`
51    #[allow(unused_variables)]
52    fn event(&self, event: Event, bounds: Bounds) {}
53
54    /// How the `View` is drawn
55    fn draw(&self, bounds: Bounds, onto: &mut impl Output);
56
57    /// # View Layout
58    /// Add padding to all sides of the `View`
59    fn padding(self, top: f32, right: f32, bottom: f32, left: f32) -> impl View {
60        Padding {
61            offsets: Offsets::new(top, right, bottom, left),
62            view: self,
63        }
64    }
65
66    /// # View Layout
67    /// Add padding to the top of the `View`
68    fn padding_top(self, pad: f32) -> impl View {
69        self.padding(pad, 0.0, 0.0, 0.0)
70    }
71
72    /// # View Layout
73    /// Add padding to the right side of the `View`
74    fn padding_right(self, pad: f32) -> impl View {
75        self.padding(0.0, pad, 0.0, 0.0)
76    }
77
78    /// # View Layout
79    /// Add padding to the bottom of the `View`
80    fn padding_bottom(self, pad: f32) -> impl View {
81        self.padding(0.0, 0.0, pad, 0.0)
82    }
83
84    /// # View Layout
85    /// Add padding to the left side of the `View`
86    fn padding_left(self, pad: f32) -> impl View {
87        self.padding(0.0, 0.0, 0.0, pad)
88    }
89
90    /// # View Layout
91    /// Add padding to the horizontal sides of the `View`
92    fn padding_horizontal(self, pad: f32) -> impl View {
93        self.padding(0.0, pad, 0.0, pad)
94    }
95
96    /// # View Layout
97    /// Add padding to the vertical sides of the `View`
98    fn padding_vertical(self, pad: f32) -> impl View {
99        self.padding(pad, 0.0, pad, 0.0)
100    }
101
102    /// # View Layout
103    /// Add different padding to the horizontal and vertical sides of the `View`
104    fn padding_both(self, x: f32, y: f32) -> impl View {
105        self.padding(y, x, y, x)
106    }
107
108    /// # View Layout
109    /// Add the same padding to all sides of the `View`
110    fn padding_all(self, pad: f32) -> impl View {
111        self.padding(pad, pad, pad, pad)
112    }
113
114    /// # View Layout
115    /// Set the size of the `View` to a fixed value.
116    fn fixed(self, width: f32, height: f32) -> impl View {
117        let size = Size::new(width, height);
118        Fixed { view: self, size }
119    }
120
121    /// # View Layout
122    /// Set the width of the `View` to a fixed value.
123    fn width(self, width: f32) -> impl View {
124        FixedWidth { view: self, width }
125    }
126
127    /// # View Layout
128    /// Set the height of the `View` to a fixed value.
129    fn height(self, height: f32) -> impl View {
130        FixedHeight { view: self, height }
131    }
132
133    /// # View Styling
134    /// Draws the `View` with the given opacity
135    /// ## Note
136    /// Opacity values range from zero to one; with
137    /// zero being fully transparent view and
138    /// one being fully opaque
139    #[track_caller]
140    fn opacity<P>(self, opacity: f32) -> Opacity<Self> {
141        let alpha = f32::clamp(opacity, 0.0, 1.0);
142        if cfg!(debug_assertions) && opacity != alpha {
143            warn_once!("opacity ({opacity}) should be between zero and one")
144        }
145
146        Opacity { view: self, alpha }
147    }
148
149    /// # View Styling
150    /// Add a background shape to the `View`; defined by a [`Path`]
151    fn background<P>(self, path: P) -> Background<Self, P> {
152        Background {
153            view: self,
154            background: path,
155        }
156    }
157
158    /// # Flexible View handling
159    /// Flexible views re-compute their size based upon the bounds of their
160    /// containing view. [`Spacer::fill`] for example.
161    #[inline(always)]
162    #[allow(unused_variables)]
163    fn adjust_width(&self, width: f32) {}
164
165    /// # Flexible View handling
166    /// Flexible views re-compute their size based upon the bounds of their
167    /// containing view. [`Spacer::fill`] for example.
168    #[inline(always)]
169    #[allow(unused_variables)]
170    fn adjust_height(&self, height: f32) {}
171
172    /// # Flexible View handling
173    /// Flexible views may be sized proportionally to each other.
174    #[doc(hidden)]
175    #[inline(always)]
176    fn frac(&self) -> usize {
177        1
178    }
179
180    /// # Gesture handling
181    /// The `View`’s response to a tap
182    fn on_tap<A, E>(self, id: Id, action: A, send: E) -> TapGesture<Self, A, E>
183    where
184        A: Clone,
185        E: Effects<Action = A>,
186    {
187        TapGesture {
188            id,
189            view: self,
190            action,
191            send,
192        }
193    }
194
195    /// # Gesture handling
196    /// The `View`’s response to a tap, but presenting a larger tap surface than is drawn
197    fn on_tap_target<A, E>(
198        self,
199        id: Id,
200        action: A,
201        send: E,
202        size: Size,
203    ) -> Target<TapGesture<Self, A, E>>
204    where
205        A: Clone,
206        E: Effects<Action = A>,
207    {
208        Target {
209            view: TapGesture {
210                id,
211                view: self,
212                action,
213                send,
214            },
215            minimum: size,
216        }
217    }
218}
