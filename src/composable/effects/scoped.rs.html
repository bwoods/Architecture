11/// An `Effects` that scopes its `Action`s to one that sends child actions.
12///
13/// This `struct` is created by the [`scope`] method on [`Effects`]. See its
14/// documentation for more.
15///
16/// [`scope`]: Effects::scope
17pub struct Scoped<Parent, Child>(pub(crate) Parent, pub(crate) PhantomData<Child>);

19impl<Parent, Child> Clone for Scoped<Parent, Child>
20where
21    Parent: Clone,
22{
23    #[inline(always)]

29impl<Parent, Child> Effects for Scoped<Parent, Child>
30where
31    <Parent as Effects>::Action: From<Child> + 'static,
32    Parent: Effects,
33    Child: 'static,
34{
35    type Action = Child;
36
37    #[inline(always)]
38    fn action(&self, action: impl Into<<Self as Effects>::Action>) {
39        self.0.action(action.into());
40    }
41
42    #[inline(always)]
43    fn stream<S: Stream<Item = <Self as Effects>::Action> + 'static>(&self, stream: S) {
44        self.0.stream(stream.map(|action| action.into()))
45    }
46
47    #[inline(always)]
48    fn future<F: Future<Output = Option<<Self as Effects>::Action>> + 'static>(&self, future: F)
49    where
50        <Self as Effects>::Action: 'static,
51    {
52        self.0
53            .future(future.map(|action| action.map(|action| action.into())))
54    }
55
56    #[inline(always)]
57    fn task<S: Stream<Item = <Self as Effects>::Action> + 'static>(&self, stream: S) -> Task {
58        self.0.task(stream.map(|action| action.into()))
59    }
60}

62#[doc(hidden)]
63impl<Parent, Child> Scheduler for Scoped<Parent, Child>
64where
65    <Parent as Effects>::Action: 'static + From<Child>,
66    Parent: Effects,
67    Child: 'static,
68{
69    type Item = Child;
70
71    fn now(&self) -> Instant {
72        self.0.now()
73    }
74
75    fn schedule_stream<
76        S: Stream<Item = ControlFlow<Instant, <Self as Scheduler>::Item>> + 'static,
77    >(
78        &self,
79        stream: S,
80    ) {
81        self.0.schedule_stream(stream.map(|action| match action {
82            ControlFlow::Continue(action) => ControlFlow::Continue(action.into()),
83            ControlFlow::Break(instant) => ControlFlow::Break(instant),
84        }))
85    }
86
87    fn schedule_task<
88        S: Stream<Item = ControlFlow<Instant, <Self as Scheduler>::Item>> + 'static,
89    >(
90        &self,
91        stream: S,
92        when: Option<Instant>,
93    ) -> Task {
94        self.0.schedule_task(
95            stream.map(|action| match action {
96                ControlFlow::Continue(action) => ControlFlow::Continue(action.into()),
97                ControlFlow::Break(instant) => ControlFlow::Break(instant),
98            }),
99            when,
100        )
101    }
