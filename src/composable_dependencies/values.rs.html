8/// A wrapper type for accessing dependencies
9pub struct Dependency<T: 'static> {
10    inner: OnceCell<Rc<T>>,
11}
12
13/// - The methods of `Dependency` are very similar to those of [`std::option::Option`], as
14///   dependencies are *optionally* present.
15/// - However, a `Dependency` on a type with a [`DependencyDefault`] also implements the
16///   [`AsRef`], [`Deref`] and [`Borrow`] traits. Even if a value has not been explicitly
17///   registered for it, the `Dependency` will still be able to [`as_ref`], [`deref`] and
18///   [`borrow`] this default value.
19///
20///  [`std::option::Option`]: std::option
21///  [`as_ref`]: Dependency::as_deref
22///  [`deref`]: Dependency::deref
23///  [`borrow`]: Dependency::borrow
24impl<T> Dependency<T> {
25    /// Retrieves the (optional) reference to the dependency of type `T`.
26    #[inline]
27    pub fn get() -> Self {
28        let cell = OnceCell::new();
29
30        if let Some(inner) = Guard::get() {
31            cell.set(inner).ok();
32        }
33
34        Self { inner: cell }
35    }
36
37    /// Returns `true` if the dependency is a [`Some`] value.
38    #[inline(always)]
39    pub fn is_some(&self) -> bool {
40        self.inner.get().is_some()
41    }
42
43    /// Returns `true` if the dependency is a [`Some`] and the value inside of it matches a predicate.
44    #[inline(always)]
45    pub fn is_some_and(&self, f: impl FnOnce(&T) -> bool) -> bool {
46        self.as_deref().filter(|inner| f(*inner)).is_some()
47    }
48
49    /// Returns `true` if the dependency is a [`None`] value.
50    #[inline(always)]
51    pub fn is_none(&self) -> bool {
52        self.inner.get().is_none()
53    }
54
55    /// Returns a slice of the dependency value, if any. If this is [`None`], an empty slice is returned.
56    #[inline(always)]
57    pub fn as_slice(&self) -> &[T] {
58        self.as_deref().map_or(&[], std::slice::from_ref)
59    }
60
61    /// Returns an iterator over the dependency value, if any.
62    #[inline(always)]
63    pub fn iter(&self) -> std::slice::Iter<'_, T> {
64        self.as_slice().iter()
65    }
66
67    /// Returns the dependency [`Some`] value.
68    ///
69    /// # Panics
70    /// Panics if the dependency is a [`None`] with a custom panic message provided by `msg`.
71    #[inline(always)]
72    pub fn expect(&self, msg: &str) -> &T {
73        self.as_deref().expect(msg)
74    }
75
76    /// Returns the contained [`Some`] value.
77    ///
78    /// # Panics
79    /// Panics if the dependency value equals [`None`].
80    #[inline(always)]
81    pub fn unwrap(&self) -> &T {
82        self.as_deref().unwrap()
83    }
84
85    /// Returns the dependency [`Some`] value or a provided default.
86    #[inline(always)]
87    pub fn unwrap_or(&self, default: T) -> Ref<'_, T> {
88        self.as_deref()
89            .map(Ref::Borrowed)
90            .unwrap_or(Ref::Owned(default))
91    }
92
93    /// Returns the dependency [`Some`] value or computes it from a closure.
94    #[inline]
95    pub fn unwrap_or_else<F>(&self, f: F) -> Ref<'_, T>
96    where
97        F: FnOnce() -> T,
98    {
99        self.as_deref()
100            .map(Ref::Borrowed)
101            .unwrap_or_else(|| Ref::Owned(f()))
102    }
103
104    /// Returns the dependency [`Some`] value or a default.
105    #[inline(always)]
106    pub fn unwrap_or_default(&self) -> Ref<'_, T>
107    where
108        T: Default,
109    {
110        self.as_deref()
111            .map(Ref::Borrowed)
112            .unwrap_or_else(|| Ref::Owned(T::default()))
113    }
114
115    /// Maps to [`Option<U>`] by applying a function to a dependency value (if [`Some`])
116    /// or returns [`None`] (if [`None`]).
117    #[inline]
118    pub fn map<U, F>(&self, f: F) -> Option<U>
119    where
120        F: FnOnce(&T) -> U,
121    {
122        self.as_deref().map(f)
123    }
124
125    /// Calls the provided closure with a reference to the dependency value (if [`Some`]).
126    #[inline]
127    pub fn inspect<F>(&self, f: F) -> Option<&T>
128    where
129        F: FnOnce(&T),
130    {
131        self.as_deref().inspect(|inner| f(*inner))
132    }
133
134    /// Returns the provided default result (if [`None`]),
135    /// or applies a function to the dependency value (if [`Some`]).
136    #[inline]
137    pub fn map_or<U, F>(&self, default: U, f: F) -> U
138    where
139        F: FnOnce(&T) -> U,
140    {
141        self.as_deref().map_or(default, f)
142    }
143
144    /// Computes a default function result (if [`None`]), or
145    /// applies a different function to the dependency value (if [`Some`]).
146    #[inline]
147    pub fn map_or_else<U, D, F>(&self, default: D, f: F) -> U
148    where
149        D: FnOnce() -> U,
150        F: FnOnce(&T) -> U,
151    {
152        self.as_deref().map_or_else(default, f)
153    }
154
155    /// Transforms into a [`Result<&T, E>`], mapping [`Some`] to
156    /// [`Ok`] and [`None`] to [`Err`].
157    #[inline(always)]
158    pub fn ok_or<E>(&self, err: E) -> Result<&T, E> {
159        self.as_deref().ok_or(err)
160    }
161
162    /// Transforms into a [`Result<&T, E>`], mapping [`Some`] to
163    /// [`Ok`] and [`None`] to [`Err`].
164    #[inline(always)]
165    pub fn ok_or_else<E, F>(&self, err: F) -> Result<&T, E>
166    where
167        F: FnOnce() -> E,
168    {
169        self.as_deref().ok_or_else(err)
170    }
171
172    /// Converts into a [`Option<&T>`].
173    ///
174    /// ## Note
175    /// This is the preferred method for producing an [`Option`] to use with the
176    /// [question mark operator][`?`].[^try]
177    ///
178    /// [`?`]: https://doc.rust-lang.org/nightly/core/option/index.html#the-question-mark-operator-
179    /// [^try]: Once the [Try trait](https://github.com/rust-lang/rust/issues/84277) is stabilized
180    ///         it will be implemented for `Dependency`.
181    #[inline]
182    pub fn as_deref(&self) -> Option<&T> {
183        self.inner.get().map(|inner| inner.deref())
184    }
185
186    /// Returns [`None`] if the dependency is [`None`], otherwise returns `rhs`.
187    #[inline(always)]
188    pub fn and<U>(&self, rhs: Option<U>) -> Option<U> {
189        self.as_deref().and(rhs)
190    }
191
192    /// Returns [`None`] if the dependency is [`None`], otherwise calls `f` with the
193    /// dependency value and returns the result.
194    #[inline]
195    pub fn and_then<U, F: FnOnce(&T) -> Option<U>>(&self, f: F) -> Option<U> {
196        self.as_deref().and_then(f)
197    }
198
199    /// Returns [`None`] if the dependency is [`None`], otherwise calls `predicate`
200    /// with the dependency value and returns:
201    /// - [`Some`] if predicate returns true (where t is the wrapped value), and
202    /// - [`None`] if predicate returns false.
203    #[inline]
204    pub fn filter<P>(&self, predicate: P) -> Option<&T>
205    where
206        P: FnOnce(&T) -> bool,
207    {
208        self.as_deref().filter(|inner| predicate(*inner))
209    }
210
211    /// Returns the dependency if it is [`Some`], otherwise returns `rhs`.
212    #[inline(always)]
213    pub fn or(&self, rhs: Option<T>) -> Option<Ref<'_, T>> {
214        self.as_deref().map(Ref::Borrowed).or(rhs.map(Ref::Owned))
215    }
216
217    /// Returns the dependency if it is [`Some`], otherwise calls `f` and returns the result.
218    #[inline]
219    pub fn or_else<F>(&self, f: F) -> Option<Ref<'_, T>>
220    where
221        F: FnOnce() -> Option<T>,
222    {
223        self.as_deref()
224            .map(Ref::Borrowed)
225            .or_else(|| f().map(Ref::Owned))
226    }
227
228    /// Returns [`Some`] if only one of
229    /// - the dependency, or
230    /// - `rhs`
231    ///
232    /// is [`Some`], otherwise returns [`None`].
233    #[inline(always)]
234    pub fn xor(&self, rhs: Option<T>) -> Option<Ref<'_, T>> {
235        self.as_deref().map(Ref::Borrowed).xor(rhs.map(Ref::Owned))
236    }
237
238    /// Maps the dependency to an [`Option<T>`] by **copying** the contents of the option.
239    #[inline(always)]
240    pub fn copied(&self) -> Option<T>
241    where
242        T: Copy,
243    {
244        self.as_deref().copied()
245    }
246
247    /// Maps the dependency to an [`Option<T>`] by **cloning** the contents of the option.
248    #[inline(always)]
249    pub fn cloned(&self) -> Option<T>
250    where
251        T: Clone,
252    {
253        self.as_deref().cloned()
254    }
255
256    /// Returns a copy of the dependency [`Some`] value or a default.
257    #[inline(always)]
258    pub fn get_or_default() -> T
259    where
260        T: Copy + Default,
261    {
262        Self::get().copied().unwrap_or_default()
263    }
264}

266/// The default value for a dependency.
267///
268/// There may be many different versions of dependencies for testing but there is often just
269/// a single default implementation for use in the the actual application.
270///
271/// Implementing this trait for a type ensures that a [`Dependency`] on it will always have
272/// a value. If the `DependencyDefault` value has not been [overridden][`super::with_dependencies`]
273/// it will be returned.
274///
275/// <div class="warning">
276/// Attempting to use this default behavior in a unit test <em>will fail the test</em>,
277/// as tests are <u>required</u> to explicitly supply all of their dependencies.
278/// </div>
279///
280/// ## Note
281/// `DependencyDefault`s are only created as needed. When its first [`Dependency`] is
282///  created, [`default`][`Default::default`] will be called once and the returned value will
283///  be cached.
284pub trait DependencyDefault: Default {}
285
286impl<T: DependencyDefault> Dependency<T> {
287    #[track_caller]
288    #[inline(never)]
289    fn get_or_insert_default(&self) -> &T {
290        self.as_deref().unwrap_or_else(|| {
291            if cfg!(test) {
292                let detailed_explanation = r#".
293
294DependencyDefault types are not allowed to use their default implementation within units tests.
295Either register the dependency on the TestStore or use with_dependency(…) within the test itself.
296"#;
297                panic!(
298                    "Dependency<{0}> was constructed during a test,\nbut {0} was not registered{1}",
299                    std::any::type_name::<T>(),
300                    detailed_explanation
301                );
302            }
303
304            let guard = Guard::new(T::default());
305            std::mem::forget(guard);
306
307            self.inner.set(Guard::get().unwrap()).ok();
308            self.as_deref().unwrap()
309        })
310    }
311}

313impl<T: DependencyDefault> Deref for Dependency<T> {

316    #[inline(always)]
317    fn deref(&self) -> &Self::Target {
318        self.get_or_insert_default()
319    }
320}
