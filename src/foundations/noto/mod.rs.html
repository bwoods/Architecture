1//! The Noto typeface families
2//!
3//! # Noto
4//! Noto is a global font collection for writing in all modern and ancient
5//! languages.
6//!
7//! Noto Serif is a modulated (“serif”) design for texts in the Latin,
8//! Cyrillic and Greek scripts, also suitable as the complementary font for
9//! other script-specific Noto Serif fonts.
10//!
11//! Noto Sans is an unmodulated (“sans serif”) design for texts in the Latin,
12//! Cyrillic and Greek scripts, which is also suitable as the complementary
13//! choice for other script-specific Noto Sans fonts.
14//!
15//! — https://fonts.google.com/noto
16#![allow(rustdoc::bare_urls)]
17#![allow(unused)]
18use composable_views::text::{Family, Font};
19
20pub struct Noto;
21
22impl Noto {
23    #[inline(never)]
24    pub fn sans(self) -> Family<'static> {
25        Font::from(include_bytes!("NotoSans-VariableFont_wdth,wght.ttf")).unwrap()
26    }
27
28    #[inline(never)]
29    pub fn serif(self) -> Family<'static> {
30        Font::from(include_bytes!("NotoSerif-VariableFont_wdth,wght.ttf")).unwrap()
31    }
32}
