5pub const fn nth_smoothed_fibonacci(n: u32) -> f32 {
6    let f = (n / 3) * 2;
7    let p = (1 << f) as f32; // ×4 for every multiple of 3
8    let q = match n % 3 {
9        0 => 1.0,
10        1 => 1.5,
11        2 => 2.5,
12        _ => unreachable!(),
13    };
14
15    p * q
16}
17
18pub const fn nth(step: u32, x: u32) -> f32 {
19    let n = match step {
20        0 => return 0.0, // zero is zero
21        _ => step - 1,   // n is base zero
22    };
23
24    nth_smoothed_fibonacci(n) * x as f32
25}
