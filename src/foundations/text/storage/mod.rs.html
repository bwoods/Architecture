11pub struct Storage {
12    characters: BTreeMap<Position, char>,
13    algorithm: Strategy,
14    clock: u32,

17impl Storage {
18    pub fn with_algorithm(algorithm: Strategy) -> Self {
19        Storage {
20            algorithm,
21            ..Default::default()
22        }
23    }
24
25    pub fn characters<R>(&self, range: R) -> impl Iterator<Item = (Position, char)>
26    where
27        R: RangeBounds<Position> + Clone,
28    {
29        // skip `Position::first()` as is it an `Exclusive` bound
30        let skip = (range.start_bound() == Unbounded) as usize;
31
32        // drop `Position::last()` as is it an `Exclusive` bound
33        let drop = (range.end_bound() == Unbounded) as usize;
34
35        self.characters
36            .range(range)
37            .dropping(skip)
38            .dropping_back(drop)
39            .map(|(pos, ch)| (pos.clone(), *ch))
40    }
41
42    pub fn replace_range<I>(&mut self, range: Range<Position>, iter: I)
43    where
44        I: Iterator<Item = char>,
45    {
46        self.remove_range(range.clone());
47        self.insert_between(range, iter)
48    }
49
50    pub fn remove_range(&mut self, range: Range<Position>) {
51        let keys = self
52            .characters
53            .range(range)
54            .map(|(pos, _)| pos.clone())
55            .collect_vec();
56
57        for key in keys {
58            self.characters.remove(&key);
59        }
60    }
61
62    pub fn remove_after(&mut self, after: Position) {
63        let range = self.range_after(after);
64
65        if range.start == Position::last() {
66            return;
67        }
68
69        self.remove_range(range);
70    }
71
72    pub fn remove_before(&mut self, before: Position) {
73        let range = self.range_before(before);
74
75        if range.start == Position::first() {
76            return;
77        }
78
79        self.remove_range(range);
80    }
81
82    pub fn insert_between<I>(&mut self, range: Range<Position>, chars: I)
83    where
84        I: Iterator<Item = char>,
85    {
86        let clock = self.next_clock();
87
88        let positions =
89            self.algorithm
90                .generate(clock, range.start.identifier(), range.end.identifier());
91
92        for (pos, ch) in positions.zip(chars) {
93            self.characters.insert(pos, ch);
94        }
95    }
96
97    pub fn insert_after<I>(&mut self, after: Position, iter: I)
98    where
99        I: Iterator<Item = char>,
100    {
101        self.insert_between(self.range_after(after), iter);
102    }
103
104    pub fn insert_before<I>(&mut self, before: Position, iter: I)
105    where
106        I: Iterator<Item = char>,
107    {
108        self.insert_between(self.range_before(before), iter);
109    }
110
111    fn range_after(&self, after: Position) -> Range<Position> {
112        let (left, right) = self
113            .characters
114            .range(after.clone()..)
115            .map(|(pos, _)| pos.clone())
116            .tuple_windows()
117            .next()
118            .unwrap(); // SAFETY: `characters` always contains STX and ETX
119
120        let right = if left == after { right } else { left };
121
122        after..right
123    }
124
125    fn range_before(&self, before: Position) -> Range<Position> {
126        let (right, left) = self
127            .characters
128            .range(..=before.clone())
129            .rev() // grab `pos` and its predecessor
130            .map(|(pos, _)| pos.clone())
131            .tuple_windows()
132            .next()
133            .unwrap(); // SAFETY: `characters` always contains STX and ETX
134
135        let left = if right == before { left } else { right };
136
137        left..before
138    }
139
140    /// The `clock` is incremented every insert to avoid the
141    /// [ABA problem](https://en.wikipedia.org/wiki/ABA_problem)
142    /// inherent in an insert-delete-insert at the same location.
143    /// # Note
144    /// The first value returned will be `1`.
145    fn next_clock(&mut self) -> u32 {
146        self.clock = u32::wrapping_add(self.clock, 1);
147        self.clock
148    }
149
150    #[cfg(test)]
151    fn string(&self) -> String {
152        self.characters(..).map(|(_, ch)| ch).collect()
153    }
154}

156impl Default for Storage {
157    fn default() -> Self {
158        let mut characters = BTreeMap::default();
159        characters.insert(Position::first(), '\u{2402}'); // ␂ (start-of-text)
160        characters.insert(Position::last(), '\u{2403}'); // ␃ (end-of-text)
161
162        Storage {
163            characters,
164            algorithm: Default::default(),
165            clock: Default::default(),
166        }
167    }
168}

170impl FromIterator<char> for Storage {
171    fn from_iter<T: IntoIterator<Item = char>>(iter: T) -> Self {
172        let mut new = Self::default();
173
174        let positions = new.algorithm.generate(new.clock, STX, ETX);
175        let chars = iter.into_iter();
176        let iter = positions.zip(chars);
177
178        new.characters.extend(iter);
179        new
180    }
181}
182
183impl Extend<char> for Storage {
184    fn extend<I: IntoIterator<Item = char>>(&mut self, iter: I) {
185        let after = self
186            .characters
187            .iter()
188            .rev() // back-to-front
189            .skip(1) // skip ETX
190            .map(|(pos, _)| pos.clone())
191            .next()
192            .unwrap(); // SAFETY: `characters` always contains STX and ETX
193
194        self.insert_after(after, iter.into_iter());
195    }
196}
